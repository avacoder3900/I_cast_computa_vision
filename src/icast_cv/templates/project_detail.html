Project API Endpoint

Python

import requests

API_URL = "{{ request.url.scheme }}://{{ request.url.netloc }}"
PROJECT_ID = "{{ project.id }}"
API_KEY = "your-api-key"

# Upload training image
with open("image.jpg", "rb") as f:
    resp = requests.post(
        f"{API_URL}/api/v1/projects/{PROJECT_ID}/training/upload",
        files={"file": f},
        data={"category": "good"},
        headers={"X-API-Key": API_KEY},
    )
print(resp.json())

# Get project stats
resp = requests.get(
    f"{API_URL}/api/v1/projects/{PROJECT_ID}/stats",
    headers={"X-API-Key": API_KEY},
)
print(resp.json())
